# Optional dependencies for the "Convert model to TensorFlow Lite" deployment
# requirement (src/export.py::export_to_tflite).
#
# These are kept separate from requirements.txt because TensorFlow + onnx2tf
# are large, slow to install, and only needed if you actually want a .tflite
# artifact (the ONNX / TorchScript exports in requirements.txt already cover
# CPU/GPU deployment without these).
#
# Install with:
#   pip install -r requirements-tflite.txt
#
# Then:
#   from src.export import export_to_tflite
#   export_to_tflite(model, dummy_input)
#
# or via the CLI pipeline:
#   python train_and_evaluate.py --tflite
#   python train_and_evaluate.py --tflite --tflite-quantize
#
# or via the running Flask app:
#   POST /api/export/tflite   {"model_id": "custom_resnet", "quantize": false}
